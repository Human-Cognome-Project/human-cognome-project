cmake_minimum_required(VERSION 3.20)
project(intent_reference LANGUAGES CXX)
enable_testing()
add_executable(intent_reference ../intent_reference.cpp)
target_compile_features(intent_reference PRIVATE cxx_std_17)
set_target_properties(intent_reference PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(intent_reference PRIVATE /W4 /WX)
else()
  target_compile_options(intent_reference PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME intent_reference COMMAND intent_reference)
