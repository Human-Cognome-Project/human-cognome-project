cmake_minimum_required(VERSION 3.18)
project(hcp_warm_toolbox LANGUAGES CXX)
find_package(PostgreSQL 16 REQUIRED)
find_package(Threads REQUIRED)
include(values.cmake)
add_library(warm_toolbox toolbox.cpp cold_reader.cpp)
target_compile_features(warm_toolbox PUBLIC cxx_std_17)
target_link_libraries(warm_toolbox PUBLIC warm_values PostgreSQL::PostgreSQL)
target_compile_options(warm_toolbox PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(warm_toolbox PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(warm_postgres_test postgres_test.cpp)
target_link_libraries(warm_postgres_test PRIVATE warm_toolbox Threads::Threads)
target_compile_options(warm_postgres_test PRIVATE -Wall -Wextra -Wpedantic)
